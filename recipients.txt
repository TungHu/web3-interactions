let lastHandledTime = 0;
let recentPrices = [];
let lastKnownPrice = null;
let open = 0.7;
let close = 0.4;
const debounceTime = 250;

const longBtn = document.querySelector("#appRoot > div.App__BodyWrapper-sc-qs2ham-1.bAGJiR > div > div.DegenMode__TradeRight-sc-5ehzd8-1.bAJJzz.scroll-style > div.DegenMode__RightBottom-sc-5ehzd8-8.HDgaQ > div > div.theme__OnePixelTop-sc-10yrp89-0.OrderBoard__BottomContent-sc-1wf193w-18.cSHbIN.eHQUTk > div > div:nth-child(1) > button > span");
const shortBtn = document.querySelector("#appRoot > div.App__BodyWrapper-sc-qs2ham-1.bAGJiR > div > div.DegenMode__TradeRight-sc-5ehzd8-1.bAJJzz.scroll-style > div.DegenMode__RightBottom-sc-5ehzd8-8.HDgaQ > div > div.theme__OnePixelTop-sc-10yrp89-0.OrderBoard__BottomContent-sc-1wf193w-18.cSHbIN.eHQUTk > div > div:nth-child(2) > button > span");

const ws = new WebSocket('wss://stream.binance.com:9443/ws/ethusdt@trade');

// 🔁 Giới hạn số lần chạy
const maxCloseActions = 10;
let closeCount = 0;

function handlePriceChange(tang, isFull) {
  const now = Date.now();
  if (now - lastHandledTime < debounceTime) return;
  lastHandledTime = now;

  const side = tang ? "Long" : "Short";
  const not = tang ? "Short" : "Long";
  let confirmed = false;
  let shouldOpenNew = isFull;

  const orders = document.querySelectorAll('.position-list .item-wrap');

  for (const order of orders) {
    const typeDiv = order.querySelector('.item-title span');
    if (!typeDiv) continue;

    const typeText = typeDiv.textContent.trim().split("·")[0];
    
    if (typeText === not) {
        const closeBtn = order.querySelector('.styles__HandleButton-sc-dki725-8');
        if (closeBtn) {
          closeBtn.click();
          closeCount++;
          console.log("⚠️ CLOSE");
		  console.log(closeCount);

          // 🛑 Nếu đạt đủ số lần thì đóng kết nối
          if (closeCount >= maxCloseActions) {
            console.log("✅ Đã đạt giới hạn CLOSE, dừng WebSocket.");
            ws.close(); // Ngừng nhận dữ liệu
          }
        } else {
          console.warn('⚠️ Không tìm thấy nút CLOSE!');
        }
    }

    if (typeText === side) {
      shouldOpenNew = false;
    }
  }

  if (shouldOpenNew) {
	if(tang ? longBtn.click() : shortBtn.click());
  }

  console.log(side, isFull);
}
 
ws.onmessage = (event) => {
  if (closeCount >= maxCloseActions) return; // Ngăn mọi xử lý sau khi đủ CLOSE

  const data = JSON.parse(event.data);
  const price = parseFloat(data.p);
  const nowTs = Date.now();
  if (isNaN(price)) return;

  recentPrices = recentPrices.filter(entry => nowTs - entry.timestamp <= 500);

  let bigMove = false;
  for (const entry of recentPrices) {
    const diff = Math.abs(price - entry.price);
    if (diff >= open - close) {
      const tang = price > entry.price;
      const isFull = diff >= open;
      bigMove = true;
      handlePriceChange(tang, isFull);
      break;
    }
  }

  if (!bigMove && recentPrices.length === 0 && lastKnownPrice !== null) {
    const delta = Math.abs(price - lastKnownPrice);
    if (delta >= open - close) {
      const tang = price > lastKnownPrice;
      const isFull = delta >= open;
      handlePriceChange(tang, isFull);
    }
  }

  recentPrices.push({ price, timestamp: nowTs });
  lastKnownPrice = price;
};
